from dotenv import load_dotenv
import os
import faostat
load_dotenv()
email = os.getenv("FAOSTAT_EMAIL")
password = os.getenv("FAOSTAT_PASSWORD")
faostat.set_requests_args(
    username=email,
    password=password,
    timeout=240)
faostat.list_datasets()
#If you get an output (a list of available datasets in a dataframe) and not an error - you are connected
